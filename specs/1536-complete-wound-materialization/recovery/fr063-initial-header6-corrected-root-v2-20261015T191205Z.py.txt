from pathlib import Path

def unique_object(pairs):
    result = {}
    for key, value in pairs:
        assert key not in result, 'Duplicate JSON key: ' + key
        result[key] = value
    return result

def loads(value):
    # AssertionError deliberately escapes ValueError/TypeError skip handlers.
    return json.loads(value, object_pairs_hook=unique_object)

def collect(value, origin, case_receipts):
    if isinstance(value, dict):
        if 'CleanupOwnedRoot' in value:
            assert value not in receipts, 'Repeated cleanup receipt is not a second qualified root'
            origins.append({'receiptIndex': len(receipts), 'origin': origin})
            receipts.append(value); case_receipts.append(value)
        for key, child in value.items():
            where = origin + '/' + key
            if key == 'Child' and isinstance(child, str):
                collect(loads(child), where + '#decoded', case_receipts)
            elif key == 'FullChildLog' and isinstance(child, str):
                for number, line in enumerate(child.splitlines(), 1):
                    try: decoded = loads(line.strip())
                    except (ValueError, TypeError):
                        assert 'CleanupOwnedRoot' not in line, 'Malformed receipt line'
                        continue
                    if isinstance(decoded, (dict, list)):
                        collect(decoded, where + '#line=' + str(number), case_receipts)
            else: collect(child, where, case_receipts)
    elif isinstance(value, list):
        for index, child in enumerate(value): collect(child, origin + '/' + str(index), case_receipts)

def one(rows, name):
    selected = [x['Observation'] for x in rows if x.get('Boundary') == name]
    assert len(selected) == 1, (name, len(selected))
    return selected[0]

def inventory(images):
    return {key: None if image is None else {'length': len(D(image)), 'sha256': H(D(image)).upper()}
        for key, image in images.items()}

def digests(images):
    return {key: {'Exists': image is not None, 'Length': None if image is None else len(D(image)),
        'Sha256': None if image is None else H(D(image)).upper()} for key, image in images.items()}

def check_dependencies(context, bindings, expected_hashes, origin):
    assert origin in ('entry', 'final')
    fields = {'Identity': 'identity', 'Path': 'path', 'Sha256': 'sha256',
        'LoadContext': 'loadContext', 'IsDefaultContext': 'isDefaultContext'}
    expected_fields = set(fields) if origin == 'entry' else set(fields.values())
    assert all(set(binding) == expected_fields for binding in bindings)
    if origin == 'entry':
        bindings = [{fields[key]: value for key, value in binding.items()} for binding in bindings]
    assert set(context['paths']) == set(context['sha256']) == {'integration', 'support', 'runtimeconfig', 'depsfile', 'immutable'}
    assert context['sha256'] == expected_hashes
    for key, path in context['paths'].items():
        assert Path(path).is_absolute() and path in fresh_by_path
        assert fresh_by_path[path]['sha256'].upper() == context['sha256'][key]
    assert len(bindings) == 3
    mapped = {'BookOfEternityClient.IntegrationTests': 'integration', 'BookOfEternityClient.TestSupport': 'support',
        'System.Collections.Immutable': 'immutable'}
    assert {x['identity'].split(',')[0] for x in bindings} == set(mapped)
    for binding in bindings:
        key = mapped[binding['identity'].split(',')[0]]
        assert binding['path'] == context['paths'][key] and binding['sha256'] == expected_hashes[key]
        assert binding['isDefaultContext'] is True and binding['loadContext'] == 'Default'
    assert next(x['identity'] for x in bindings if x['identity'].startswith('System.Collections.Immutable,')) == context['immutableAssemblyIdentity']
