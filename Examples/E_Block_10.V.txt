   <title>
    Mounting a Horse - activeVehicleChange Command
   </title>
   <content language="json" type="code_example">
    <![CDATA[
    // Example of a player mounting their horse
    "activeVehicleChange": "horse-destrier-thunderhoof-001",
    
    // Corresponding vehicle that becomes active (already exists in playerVehicles)
    "UpdateVehicles": [
        {
            "vehicleId": "horse-destrier-thunderhoof-001",
            "availability": "Active"
        }
    ]
    ]]>
   </content>
  </example>

   </title>
   <content language="json" type="code_example">

            "vehicleId": "horse-destrier-thunderhoof-001",
